# 固件验证流程实现

> Bootloader 收到 [[8. 完整固件包格式设计|固件包]] 后的完整验证链路：魔数 → 版本号（防降级）→ 头部 CRC → AES 解密 → SHA256 完整性 → ECDSA 验签 → 写 Flash → 跳转 App。验证顺序由轻到重，越早失败越省 CPU。详见 [[1. OTA 固件升级完整流程|总览]] 的设备端验证流程。

---

## 完整实现代码

### firmware_pack.h

```c
#ifndef FIRMWARE_PACK_H
#define FIRMWARE_PACK_H

#include <stdint.h>
#include "mbedtls/aes.h"
#include "mbedtls/sha256.h"
#include "mbedtls/ecdsa.h"
#include "mbedtls/ecp.h"

// ===== Flash 地址配置（❗ 必须与链接脚本一致）=====
#define APP_FLASH_ADDR       0x08008000  // App 区起始地址
#define PARAM_FLASH_ADDR     0x08004000  // 参数区起始地址（存版本号）
#define APP_MAX_SIZE        (256 * 1024) // App 区最大大小

// ===== AES 密钥（❗ 必须与 PC 打包脚本一致）=====
static const uint8_t AES_KEY[16] = {
    0x01, 0x02, 0x03, 0x04, 0x05, 0x06, 0x07, 0x08,
    0x09, 0x0A, 0x0B, 0x0C, 0x0D, 0x0E, 0x0F, 0x10
};

// ===== ECC 公钥（编译时硬编码，对应 PC 端私钥）=====
// 用 openssl ec -in ec_private.pem -pubout -out ec_public.pem 生成后提取
static const uint8_t ECC_PUBLIC_KEY[] = {
    // 04 开头表示非压缩格式，后跟 x(32字节) + y(32字节) = 65字节
    0x04,
    // ❗ 替换为你的实际公钥字节
    0x00, 0x00, /* ... 32字节 x 坐标 ... */
    0x00, 0x00  /* ... 32字节 y 坐标 ... */
};

// 固件头结构（严格对应 PC 打包脚本，128字节）
typedef struct {
    uint8_t  magic[4];       // "FWPK"
    uint32_t version;        // 固件版本号
    uint32_t firmware_size;  // 原始固件大小（PKCS7填充前）
    uint8_t  iv[16];         // AES-CBC IV
    uint8_t  sha256[32];     // 原始固件的 SHA256
    uint8_t  signature[64];  // ECDSA-P256 签名（r=32 + s=32）
    uint32_t header_crc;     // 头部 CRC32
} __attribute__((packed)) FirmwareHeader_t;

// 编译期断言，确保结构体大小正确
_Static_assert(sizeof(FirmwareHeader_t) == 128, "FirmwareHeader size must be 128 bytes");

int firmware_unpack_and_verify(uint8_t *pkg_buf, uint32_t pkg_size);

#endif
```

### firmware_pack.c

```c
#include "firmware_pack.h"
#include "flash.h"       // flash_erase_sectors_by_addr / flash_write_word
#include "crc32.h"       // crc32_calc

// 解密缓冲区（静态分配，避免大块栈分配导致栈溢出）
// ⚠️ 大小必须 >= App 最大固件大小 + 16字节（PKCS7最大填充）
static uint8_t decrypted_buf[APP_MAX_SIZE + 16];

int firmware_unpack_and_verify(uint8_t *pkg_buf, uint32_t pkg_size)
{
    // ===== 第一关：基本长度检查 =====
    if (pkg_size < sizeof(FirmwareHeader_t)) {
        printf("[OTA] 固件包太小：%lu 字节\r\n", pkg_size);
        return -1;
    }

    FirmwareHeader_t *header  = (FirmwareHeader_t *)pkg_buf;
    uint8_t          *cipher  = pkg_buf + sizeof(FirmwareHeader_t);
    uint32_t          cipher_size = pkg_size - sizeof(FirmwareHeader_t);

    // ===== 第二关：魔数检查（最轻量，4字节比较）=====
    if (memcmp(header->magic, "FWPK", 4) != 0) {
        printf("[OTA] 魔数错误，非法固件包\r\n");
        return -2;
    }

    // ===== 第三关：版本号检查（防降级）=====
    uint32_t current_ver = *(volatile uint32_t *)PARAM_FLASH_ADDR;
    if (current_ver != 0xFFFFFFFF && header->version <= current_ver) {
        // 0xFFFFFFFF 说明参数区未初始化（首次烧录），跳过版本检查
        printf("[OTA] 拒绝降级：当前版本 %lu，新版本 %lu\r\n",
               current_ver, header->version);
        return -3;
    }

    // ===== 第四关：头部 CRC 验证（防头部损坏）=====
    uint32_t crc_calc = crc32_calc((uint8_t *)header,
                                    sizeof(FirmwareHeader_t) - 4);
    if (crc_calc != header->header_crc) {
        printf("[OTA] 头部 CRC 错误：计算 0x%08lX，期望 0x%08lX\r\n",
               crc_calc, header->header_crc);
        return -4;
    }

    // ===== 第五关：固件大小合法性检查 =====
    if (header->firmware_size == 0 ||
        header->firmware_size > APP_MAX_SIZE) {
        printf("[OTA] 固件大小非法：%lu 字节\r\n", header->firmware_size);
        return -5;
    }

    // ===== 第六关：AES-CBC 解密（计算量较大）=====
    uint8_t iv_copy[16];
    memcpy(iv_copy, header->iv, 16);  // mbedtls 会修改 iv，必须备份

    mbedtls_aes_context aes;
    mbedtls_aes_init(&aes);
    int ret = mbedtls_aes_setkey_dec(&aes, AES_KEY, 128);
    if (ret != 0) {
        mbedtls_aes_free(&aes);
        printf("[OTA] AES 密钥设置失败：%d\r\n", ret);
        return -6;
    }

    ret = mbedtls_aes_crypt_cbc(&aes,
        MBEDTLS_AES_DECRYPT,
        cipher_size,    // 必须是16的整数倍
        iv_copy,
        cipher,
        decrypted_buf);
    mbedtls_aes_free(&aes);

    if (ret != 0) {
        printf("[OTA] AES 解密失败：%d\r\n", ret);
        return -7;
    }

    // ===== 第七关：SHA256 完整性校验 =====
    // ⚠️ 用 firmware_size 截断，去掉 PKCS7 填充字节
    uint8_t digest[32];
    mbedtls_sha256_context sha;
    mbedtls_sha256_init(&sha);
    mbedtls_sha256_starts(&sha, 0);  // 0 = SHA256，1 = SHA224
    mbedtls_sha256_update(&sha, decrypted_buf, header->firmware_size);
    mbedtls_sha256_finish(&sha, digest);
    mbedtls_sha256_free(&sha);

    if (memcmp(digest, header->sha256, 32) != 0) {
        printf("[OTA] SHA256 不匹配，固件损坏或被篡改\r\n");
        return -8;
    }

    // ===== 第八关：ECDSA 签名验证（最重量，椭圆曲线运算）=====
    // ⚠️ ECDSA 与 RSA 的关键区别：
    //    RSA 验证可以从签名里"解出"哈希值，hash 是输出
    //    ECDSA 验证必须传入 hash，hash 是数学运算的输入
    //    这里的 header->sha256 不是"用来比对的"，是参与 u1*G + u2*Q 计算的核心参数
    //    详见：5. ECDSA数字签名原理（解决真实性）
    mbedtls_ecdsa_context ecdsa;
    mbedtls_ecdsa_init(&ecdsa);

    ret = mbedtls_ecp_group_load(&ecdsa.grp, MBEDTLS_ECP_DP_SECP256R1);
    if (ret == 0) {
        ret = mbedtls_ecp_point_read_binary(&ecdsa.grp, &ecdsa.Q,
                  ECC_PUBLIC_KEY, sizeof(ECC_PUBLIC_KEY));
    }
    if (ret == 0) {
        // signature[64] = r(32字节) + s(32字节) 拼接格式
        ret = mbedtls_ecdsa_read_signature(&ecdsa,
                  header->sha256, 32,
                  header->signature, 64);
    }
    mbedtls_ecdsa_free(&ecdsa);

    if (ret != 0) {
        printf("[OTA] ECDSA 验签失败，固件来源不可信：%d\r\n", ret);
        return -9;
    }

    // ===== 全部验证通过，写入 Flash =====
    printf("[OTA] 验证全部通过，开始擦除并写入 Flash...\r\n");

    HAL_FLASH_Unlock();

    // 擦除 App 区对应的扇区
    flash_erase_sectors_by_addr(APP_FLASH_ADDR, header->firmware_size);

    // 按4字节逐字写入，写完立刻回读验证
    for (uint32_t i = 0; i < header->firmware_size; i += 4) {
        uint32_t word = *(uint32_t *)(decrypted_buf + i);
        HAL_FLASH_Program(FLASH_TYPEPROGRAM_WORD, APP_FLASH_ADDR + i, word);

        if (*(volatile uint32_t *)(APP_FLASH_ADDR + i) != word) {
            HAL_FLASH_Lock();
            printf("[OTA] Flash 写入验证失败，地址 0x%08lX\r\n",
                   APP_FLASH_ADDR + i);
            return -10;
        }
    }

    // 更新版本号到参数区
    HAL_FLASH_Program(FLASH_TYPEPROGRAM_WORD,
                      PARAM_FLASH_ADDR,
                      header->version);

    HAL_FLASH_Lock();

    printf("[OTA] 写入完成，固件版本更新为 %lu\r\n", header->version);
    return 0;  // 成功
}
```

## 整合到 Bootloader 主流程

```c
// bootloader_main.c

// Ymodem 接收缓冲区（头部128字节 + 加密固件，需足够大）
// ⚠️ 注意 RAM 用量：接收缓冲区 + 解密缓冲区 = 约 2 × App最大大小
static uint8_t pkg_recv_buf[APP_MAX_SIZE + 256];  // 留256字节余量

void bootloader_ota_task(void)
{
    printf("[OTA] 等待固件包（Ymodem）...\r\n");

    // Ymodem 接收固件包到缓冲区，返回实际接收的字节数
    int32_t pkg_size = Ymodem_Receive(pkg_recv_buf);

    if (pkg_size <= 0) {
        printf("[OTA] Ymodem 接收失败：%ld\r\n", pkg_size);
        return;
    }

    printf("[OTA] 接收完成，大小：%ld 字节，开始验证...\r\n", pkg_size);

    // 解包、验证、写 Flash
    int ret = firmware_unpack_and_verify(pkg_recv_buf, (uint32_t)pkg_size);

    if (ret == 0) {
        printf("[OTA] 升级成功！3秒后跳转 App\r\n");
        HAL_Delay(3000);
        jump_to_app(APP_FLASH_ADDR);
    } else {
        printf("[OTA] 升级失败，错误码：%d，保持旧固件运行\r\n", ret);
        // 不跳转，停在 Bootloader 等待重试或人工干预
    }
}
```

## 错误码速查

| 返回值 | 含义 | 排查方向 |
|--------|------|---------|
| -1 | 固件包长度小于头部大小 | 传输不完整，检查 Ymodem 接收逻辑 |
| -2 | 魔数错误 | 发送的不是固件包，或头部损坏 |
| -3 | 版本号降级被拒 | 打包脚本里的 FIRMWARE_VERSION 没有递增 |
| -4 | 头部 CRC 错误 | 传输中头部被篡改或损坏 |
| -5 | 固件大小非法 | firmware_size 字段异常，头部已损坏 |
| -6 | AES 密钥设置失败 | mbedTLS 配置问题 |
| -7 | AES 解密失败 | cipher_size 不是16的整数倍 |
| -8 | SHA256 不匹配 | 固件内容被篡改，或 AES 密钥/IV 不一致 |
| -9 | ECDSA 验签失败 | 私钥和公钥不匹配，或签名字段损坏 |
| -10 | Flash 写入验证失败 | Flash 硬件问题，或电压不稳 |

## 版本号存储位置的选择

| 位置 | 优点 | 缺点 |
|------|------|------|
| Flash 参数区（推荐） | 掉电不丢，擦写寿命够用 | 需要单独管理扇区 |
| RTC 备份寄存器 | 读写简单 | 完全掉电（VBAT 也没电）时会丢失 |
| OTP 区域 | 不可篡改 | 只能写一次，无法持续更新版本号 |

> ⚠️ 不能用 RTC 备份寄存器存版本号，因为攻击者可以故意让 VBAT 掉电来绕过防降级检查。
