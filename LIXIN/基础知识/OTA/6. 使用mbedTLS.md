# 使用 mbedTLS 实现加解密与验签

> mbedTLS 是 ARM 官方的轻量级 TLS/加密库，专为嵌入式设计。在 OTA 中用于 STM32 端的所有密码学操作。PC 端对应的工具是 [[7. 使用OpenSSL命令|OpenSSL]]。

---

## AES-CBC 加解密

```c
#include "mbedtls/aes.h"

// 密钥和 IV（实际项目中密钥存 OTP，IV 随机生成每次不同）
uint8_t key[16] = {0x01,0x02,...};  // 128bit 密钥
uint8_t iv[16]  = {0x00,0x00,...};  // 初始向量
uint8_t iv_copy[16];                // CBC 加密会修改 iv，备份一份

// 加密
mbedtls_aes_context aes;
mbedtls_aes_init(&aes);
mbedtls_aes_setkey_enc(&aes, key, 128);

memcpy(iv_copy, iv, 16);  // 备份 iv
mbedtls_aes_crypt_cbc(&aes,
    MBEDTLS_AES_ENCRYPT,
    plaintext_len,   // 必须是16的整数倍
    iv_copy,         // 加密后 iv_copy 会被修改为最后一块密文
    plaintext,
    ciphertext);

mbedtls_aes_free(&aes);

// 解密
mbedtls_aes_setkey_dec(&aes, key, 128);
memcpy(iv_copy, iv, 16);  // 用原始 iv
mbedtls_aes_crypt_cbc(&aes,
    MBEDTLS_AES_DECRYPT,
    ciphertext_len,
    iv_copy,
    ciphertext,
    plaintext);
```

## SHA256 计算

```c
#include "mbedtls/sha256.h"

uint8_t digest[32];  // SHA256 输出 32 字节

mbedtls_sha256_context sha;
mbedtls_sha256_init(&sha);
mbedtls_sha256_starts(&sha, 0);  // 0=SHA256, 1=SHA224

// 可以分多次喂数据，适合大文件分块处理
mbedtls_sha256_update(&sha, data, data_len);
// 如果固件很大，分块：
// while (收到一块数据) { mbedtls_sha256_update(&sha, block, block_size); }

mbedtls_sha256_finish(&sha, digest);
mbedtls_sha256_free(&sha);

// digest 现在是 32 字节的 SHA256 摘要
```

## ECC 签名验证

```c
#include "mbedtls/ecdsa.h"
#include "mbedtls/ecp.h"

// 验证签名（STM32 端只需要验证，不需要签名）
mbedtls_ecdsa_context ecdsa;
mbedtls_ecdsa_init(&ecdsa);

// 加载内置公钥（编译时烧进去的）
mbedtls_ecp_group_load(&ecdsa.grp, MBEDTLS_ECP_DP_SECP256R1);
mbedtls_ecp_point_read_binary(&ecdsa.grp, &ecdsa.Q,
    public_key_bytes, public_key_len);

// 验证签名
int ret = mbedtls_ecdsa_read_signature(&ecdsa,
    digest,     // SHA256 摘要
    32,
    signature,  // 收到的签名
    signature_len);

if (ret == 0) {
    // 签名有效，固件可信
} else {
    // 签名无效，拒绝
}

mbedtls_ecdsa_free(&ecdsa);
```
